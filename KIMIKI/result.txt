const canvas = document.getElementById('canvas') as HTMLCanvasElement;
const ctx = canvas.getContext('2d')!;

let width = window.innerWidth;
let height = window.innerHeight;
canvas.width = width;
canvas.height = height;

window.addEventListener('resize', () => {
  width = window.innerWidth;
  height = window.innerHeight;
  canvas.width = width;
  canvas.height = height;
});

interface Vec2 {
  x: number;
  y: number;
}

const ACCEL = 450; // px/s²
const MAX_CIRCLES = 150;

function dist(a: Vec2, b: Vec2): number {
  const dx = a.x - b.x;
  const dy = a.y - b.y;
  return Math.sqrt(dx * dx + dy * dy);
}

function mag(v: Vec2): number {
  return Math.sqrt(v.x * v.x + v.y * v.y);
}

function normalize(v: Vec2): Vec2 {
  const m = mag(v);
  if (m === 0) return { x: 0, y: 0 };
  return { x: v.x / m, y: v.y / m };
}

function sub(a: Vec2, b: Vec2): Vec2 {
  return { x: a.x - b.x, y: a.y - b.y };
}

function randomPoint(from: Vec2, minDist: number, maxDist: number): Vec2 {
  for (let i = 0; i < 20; i++) {
    const angle = Math.random() * Math.PI * 2;
    const d = minDist + Math.random() * (maxDist - minDist);
    const p: Vec2 = {
      x: from.x + Math.cos(angle) * d,
      y: from.y + Math.sin(angle) * d,
    };
    const margin = 50;
    if (p.x > margin && p.x < width - margin && p.y > margin && p.y < height - margin) {
      return p;
    }
  }
  const angle = Math.random() * Math.PI * 2;
  const d = minDist + Math.random() * (maxDist - minDist);
  return {
    x: Math.max(50, Math.min(width - 50, from.x + Math.cos(angle) * d)),
    y: Math.max(50, Math.min(height - 50, from.y + Math.sin(angle) * d)),
  };
}

class Circle {
  origin: Vec2;
  pos: Vec2;
  target: Vec2;
  vel: Vec2;
  hue: number;

  constructor(origin: Vec2, target: Vec2) {
    this.origin = { ...origin };
    this.pos = { ...origin };
    this.target = { ...target };
    this.vel = { x: 0, y: 0 };
    this.hue = Math.random() * 360;
  }

  setSegment(origin: Vec2, target: Vec2) {
    this.origin = { ...origin };
    this.pos = { ...origin };
    this.target = { ...target };
    this.vel = { x: 0, y: 0 };
  }

  update(dt: number): boolean {
    const toTarget = sub(this.target, this.pos);
    const d = mag(toTarget);
    if (d < 0.5) {
      this.pos = { ...this.target };
      return true;
    }

    const dir = normalize(toTarget);
    this.vel.x += dir.x * ACCEL * dt;
    this.vel.y += dir.y * ACCEL * dt;

    const speed = mag(this.vel);
    if (speed * dt >= d) {
      this.pos = { ...this.target };
      return true;
    }

    this.pos.x += this.vel.x * dt;
    this.pos.y += this.vel.y * dt;
    return false;
  }
}

const circles: Circle[] = [];
let paused = false;

function init() {
  const center: Vec2 = { x: width / 2, y: height / 2 };
  const firstTarget = randomPoint(center, 500, 900);
  circles.push(new Circle(center, firstTarget));
}

function spawnChild(origin: Vec2) {
  const target = randomPoint(origin, 500, 900);
  circles.push(new Circle(origin, target));
  if (circles.length > MAX_CIRCLES) {
    circles.shift();
  }
}

let lastTime = performance.now();

function loop(now: number) {
  requestAnimationFrame(loop);
  const rawDt = (now - lastTime) / 1000;
  lastTime = now;
  const dt = Math.min(rawDt, 0.05);

  if (!paused) {
    for (let i = circles.length - 1; i >= 0; i--) {
      const c = circles[i];
      const arrived = c.update(dt);
      if (arrived) {
        const oldOrigin = { ...c.origin };
        const newTarget = randomPoint(c.target, 500, 900);
        c.setSegment(c.target, newTarget);
        spawnChild(oldOrigin);
      }
    }
  }

  ctx.fillStyle = '#0f172a';
  ctx.fillRect(0, 0, width, height);

  for (const c of circles) {
    const speed = mag(c.vel);
    const lightness = 30 + Math.min(speed / 8, 50);
    const radius = 4 + Math.min(speed / 40, 8);

    ctx.beginPath();
    ctx.arc(c.pos.x, c.pos.y, radius, 0, Math.PI * 2);
    ctx.fillStyle = `hsl(${c.hue} 80% ${lightness}%)`;
    ctx.fill();
  }
}

window.addEventListener('keydown', (e) => {
  if (e.code === 'Space') {
    e.preventDefault();
    paused = !paused;
  }
});

init();
requestAnimationFrame(loop);

Design choices
Acceleration: Fixed at 450 px/s². Velocity resets to 0 at the start of every segment so each leg truly “starts slow.”
Arrival detection: If the distance remaining in a frame is smaller than the step that frame would cover, the circle snaps to its target to avoid overshoot.
Spawn cap: Hard limit of 150 circles keeps the simulation performant as the branching process grows.
Color: Each circle gets a random hue; lightness maps from 30 % (slow) up to 80 % (fast). Radius also grows slightly with speed (4–12 px).
Viewport bounds: New targets retry up to 20 random angles to land inside a 50 px padded viewport; if that fails, the point is clamped to remain on-screen.
Pause: Spacebar toggles the animation loop; preventDefault stops the page from scrolling.
