```typescript
// ── Canvas setup ──────────────────────────────────────
const canvas = document.getElementById('canvas') as HTMLCanvasElement;
const ctx = canvas.getContext('2d')!;
const pauseEl = document.getElementById('pause-overlay') as HTMLDivElement;

let W = 0;
let H = 0;

function resize(): void {
  W = window.innerWidth;
  H = window.innerHeight;
  canvas.width = W;
  canvas.height = H;
}
resize();
window.addEventListener('resize', resize);

// ── Types ────────────────────────────────────────────
interface Circle {
  sx: number;  // start x
  sy: number;  // start y
  tx: number;  // target x
  ty: number;  // target y
  dx: number;  // unit direction x
  dy: number;  // unit direction y
  d: number;   // total distance to target
  s: number;   // displacement traveled so far
  hue: number;
}

interface Flash {
  x: number;
  y: number;
  hue: number;
  t: number;   // normalised age 0 → 1
}

// ── Constants ────────────────────────────────────────
const ACCEL       = 300;   // px/s²  constant acceleration
const DIST_MIN    = 500;   // minimum travel distance
const DIST_MAX    = 900;   // maximum travel distance
const RADIUS      = 10;    // core circle radius
const MAX_CIRCLES = 200;   // hard cap to keep things smooth
const FLASH_DUR   = 0.4;   // seconds a flash ring lasts

// ── State ────────────────────────────────────────────
let circles: Circle[] = [];
let flashes: Flash[]  = [];
let paused  = false;
let hueIdx  = Math.random() * 360;  // start at a random hue

// ── Helpers ──────────────────────────────────────────

/** Golden-angle hue stepper — keeps colours well-spread. */
function nextHue(): number {
  hueIdx = (hueIdx + 137.508) % 360;
  return hueIdx;
}

function randDist(): number {
  return DIST_MIN + Math.random() * (DIST_MAX - DIST_MIN);
}

/** Create a new circle that will accelerate from (sx,sy) toward a random target. */
function spawn(sx: number, sy: number): Circle {
  const dist  = randDist();
  const angle = Math.random() * Math.PI * 2;
  return {
    sx, sy,
    tx: sx + Math.cos(angle) * dist,
    ty: sy + Math.sin(angle) * dist,
    dx: Math.cos(angle),
    dy: Math.sin(angle),
    d: dist,
    s: 0,
    hue: nextHue(),
  };
}

/** Instantaneous speed at current displacement (kinematic eq: v² = 2as). */
function speed(c: Circle): number {
  return Math.sqrt(Math.max(0, 2 * ACCEL * c.s));
}

/** Speed the circle will have exactly at the target. */
function maxSpeed(c: Circle): number {
  return Math.sqrt(Math.max(0, 2 * ACCEL * c.d));
}

/** World position from start + displacement along direction. */
function pos(c: Circle): [number, number] {
  return [c.sx + c.dx * c.s, c.sy + c.dy * c.s];
}

// ── Initialise first circle at screen centre ─────────
circles.push(spawn(W / 2, H / 2));

// ── Input ────────────────────────────────────────────
window.addEventListener('keydown', (e: KeyboardEvent) => {
  if (e.code === 'Space') {
    e.preventDefault();
    paused = !paused;
    if (!paused) lastT = performance.now();   // avoid a huge dt jump
    pauseEl.classList.toggle('visible', paused);
  }
});

// ── Physics update ───────────────────────────────────
function update(dt: number): void {
  const born: Circle[] = [];

  for (const c of circles) {
    // Kinematic update:  Δs = v·Δt + ½·a·Δt²
    const v = speed(c);
    c.s += v * dt + 0.5 * ACCEL * dt * dt;

    // Check arrival
    if (c.s >= c.d) {
      c.s = c.d;  // clamp

      // Visual flash at arrival point
      flashes.push({ x: c.tx, y: c.ty, hue: c.hue, t: 0 });

      // Remember old start for the branch
      const ox = c.sx;
      const oy = c.sy;

      // "B becomes the new A" — reset this circle to continue from target
      const dist  = randDist();
      const angle = Math.random() * Math.PI * 2;
      c.sx  = c.tx;
      c.sy  = c.ty;
      c.tx  = c.sx + Math.cos(angle) * dist;
      c.ty  = c.sy + Math.sin(angle) * dist;
      c.dx  = Math.cos(angle);
      c.dy  = Math.sin(angle);
      c.d   = dist;
      c.s   = 0;
      c.hue = nextHue();

      // "Simultaneously a new circle spawns from the previous A"
      if (circles.length + born.length < MAX_CIRCLES) {
        born.push(spawn(ox, oy));
      }
    }
  }

  circles.push(...born);

  // Age and cull flashes
  for (const f of flashes) f.t += dt / FLASH_DUR;
  flashes = flashes.filter(f => f.t < 1);
}

// ── Render ───────────────────────────────────────────
function draw(): void {
  // Full clear
  ctx.fillStyle = '#08080d';
  ctx.fillRect(0, 0, W, H);

  // ─ Flash rings ─
  for (const f of flashes) {
    const r = RADIUS + f.t * 65;
    const alpha = (1 - f.t) * 0.5;
    ctx.strokeStyle = `hsla(${f.hue},100%,65%,${alpha})`;
    ctx.lineWidth = 2.5 * (1 - f.t);
    ctx.beginPath();
    ctx.arc(f.x, f.y, r, 0, Math.PI * 2);
    ctx.stroke();
  }

  // ─ Circles ─
  for (const c of circles) {
    const [x, y] = pos(c);
    const vm = maxSpeed(c);
    const t  = vm > 0 ? Math.min(speed(c) / vm, 1) : 0;
    const l  = 18 + t * 58;               // lightness 18 % → 76 %
    const gr = RADIUS * (2.5 + t * 4.5);   // glow radius grows with speed

    // Soft radial glow
    const g = ctx.createRadialGradient(x, y, 0, x, y, gr);
    g.addColorStop(0,   `hsla(${c.hue},100%,${l}%,0.7)`);
    g.addColorStop(0.3, `hsla(${c.hue},90%,${l}%,0.18)`);
    g.addColorStop(1,   `hsla(${c.hue},80%,${l}%,0)`);
    ctx.fillStyle = g;
    ctx.beginPath();
    ctx.arc(x, y, gr, 0, Math.PI * 2);
    ctx.fill();

    // Solid core (slightly brighter than the glow centre)
    ctx.fillStyle = `hsl(${c.hue},100%,${Math.min(l + 18, 93)}%)`;
    ctx.beginPath();
    ctx.arc(x, y, RADIUS, 0, Math.PI * 2);
    ctx.fill();
  }

  // ─ HUD ─
  ctx.fillStyle = '#08080d';
  ctx.fillRect(0, 0, 170, 54);
  ctx.fillStyle = 'rgba(255,255,255,0.28)';
  ctx.font = '12px monospace';
  ctx.textBaseline = 'top';
  ctx.fillText(`circles: ${circles.length}`, 10, 10);
  ctx.fillText('space to pause', 10, 28);
}

// ── Main loop ────────────────────────────────────────
let lastT = performance.now();

function frame(now: number): void {
  requestAnimationFrame(frame);
  const dt = Math.min((now - lastT) / 1000, 0.05); // cap to avoid spiral-of-death
  lastT = now;
  if (!paused) update(dt);
  draw();
}

requestAnimationFrame(frame);
```

---

**Design choices noted:**

- **Acceleration 300 px/s²** — gives ~1.8–2.5 s travel times for the 500–900 px range, feeling brisk but visible.
- **Golden-angle hue stepping (137.508°)** — distributes colours evenly around the wheel without repetition.
- **HSL lightness 18 %→76 %** mapped to speed ratio — circles start dim and brighten noticeably as they accelerate.
- **Radial-gradient glow whose radius scales with speed** — reinforces the brightening effect visually.
- **Expanding ring flash on arrival** — gives clear feedback when a circle reaches its target and spawns.
- **Hard cap of 200 circles** — prevents exponential branch growth from tanking frame rate; oldest branches simply stop spawning once the cap is hit.
- **`lastTime` reset on unpause** — avoids a single massive `dt` spike after a long pause.
- **`dt` capped at 50 ms** — guards against tab-background throttling causing physics explosions.
